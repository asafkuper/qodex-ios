func lifePath(birthDate: Date) -> Int {
    let components = calendar.dateComponents([.day, .month, .year], from: birthDate)
    let day = reduceToSingleDigit(components.day!)
    let month = reduceToSingleDigit(components.month!)
    let year = reduceToSingleDigit(components.year!)
